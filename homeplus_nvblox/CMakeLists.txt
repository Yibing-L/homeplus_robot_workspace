cmake_minimum_required(VERSION 3.22)
project(homeplus_nvblox LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
foreach(dep rclcpp sensor_msgs std_msgs std_srvs shape_msgs visualization_msgs
    moveit_msgs moveit_ros_planning geometric_shapes tf2_ros tf2_eigen)
  find_package(${dep} REQUIRED)
endforeach()
find_package(CUDAToolkit REQUIRED)
# scripts/setup_nvblox.sh installs the pinned upstream library here.
get_filename_component(NVBLOX_LOCAL_PREFIX "${CMAKE_CURRENT_SOURCE_DIR}/../.deps/nvblox" ABSOLUTE)
list(PREPEND CMAKE_PREFIX_PATH "${NVBLOX_LOCAL_PREFIX}")
set(GFLAGS_USE_TARGET_NAMESPACE FALSE)
find_package(gflags REQUIRED CONFIG PATHS "${NVBLOX_LOCAL_PREFIX}/lib/cmake/gflags" NO_DEFAULT_PATH)
find_package(glog REQUIRED CONFIG PATHS "${NVBLOX_LOCAL_PREFIX}/lib/cmake/glog" NO_DEFAULT_PATH)
find_package(nvblox 0.0.10 EXACT REQUIRED CONFIG)
add_executable(nvblox_mapper src/nvblox_mapper.cpp)
target_compile_features(nvblox_mapper PRIVATE cxx_std_17)
ament_target_dependencies(nvblox_mapper rclcpp sensor_msgs std_msgs std_srvs
  shape_msgs visualization_msgs moveit_msgs moveit_ros_planning geometric_shapes tf2_ros tf2_eigen)
target_link_libraries(nvblox_mapper nvblox::nvblox_lib CUDA::cudart)
set_target_properties(nvblox_mapper PROPERTIES INSTALL_RPATH "${NVBLOX_LOCAL_PREFIX}/lib")
install(TARGETS nvblox_mapper DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
ament_package()
